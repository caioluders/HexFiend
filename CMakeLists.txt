cmake_minimum_required(VERSION 3.20)

project(HexFiend LANGUAGES C CXX)

option(HEXFIEND_BUILD_CORE "Build the portable HexFiendCore library" ON)
option(HEXFIEND_BUILD_LINUX_APP "Build the Linux ImGui application" ON)

include(CTest)

if(HEXFIEND_BUILD_CORE)
    if(NOT DEFINED CMAKE_OBJC_COMPILER)
        find_program(HEXFIEND_CLANG clang)
        if(HEXFIEND_CLANG)
            set(CMAKE_OBJC_COMPILER "${HEXFIEND_CLANG}" CACHE FILEPATH "Objective-C compiler" FORCE)
        endif()
    endif()
    if(NOT DEFINED CMAKE_OBJCXX_COMPILER)
        find_program(HEXFIEND_CLANGXX clang++)
        if(HEXFIEND_CLANGXX)
            set(CMAKE_OBJCXX_COMPILER "${HEXFIEND_CLANGXX}" CACHE FILEPATH "Objective-C++ compiler" FORCE)
        endif()
    endif()
    enable_language(OBJC OPTIONAL)
    add_subdirectory(core)
else()
    message(STATUS "HexFiendCore target disabled.")
endif()

if(HEXFIEND_BUILD_LINUX_APP AND TARGET HexFiend::Core AND UNIX AND NOT APPLE)
    enable_language(OBJCXX OPTIONAL)
    add_subdirectory(ui/linux)
elseif(HEXFIEND_BUILD_LINUX_APP)
    message(STATUS "HexFiend Linux app target disabled: requires Unix, Objective-C++, and HexFiendCore.")
endif()
